## Build Script Template for Habitat Construction
```csharp
// Habitat Building System with Progress Tracking
public class HabitatBuilder : MonoBehaviour {
    public float constructionProgress = 0f;
    public Material[] buildingMaterials = new Material[5];
    
    void OnEnable() {
        DontDestroyOnLoad(gameObject);
        StartCoroutine(BuildHabitatSequence());
    }
    
    IEnumerator BuildHabitatSequence() {
        yield return new WaitForSeconds(2f); // Foundation delay
        constructionProgress += 10f;
        
        yield return new WaitForSeconds(3f); // Frame assembly delay
        constructionProgress += 25f;
        
        // Add warning if progress exceeds target
        if (constructionProgress >= 90) {
            Debug.Log("WARNING: Structure near completion");
        }
    }
}
```
